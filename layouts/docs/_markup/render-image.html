{{- /* Image render hook. - A remote image is fetched at build time (resources.GetRemote; the allowed hosts are in config/_default/security.toml) and served from this site. - A local SVG from assets/ is put inline. A remote SVG is never put inline: its markup comes from another host and could carry scripts, so it is shown with an element, where scripts do not run. - If the image cannot be found, the destination is used as it is; Hugo still checks the URL, so "javascript:" and the like are replaced. */ -}} {{- $dest := .Destination -}} {{- $text := .PlainText -}} {{- $url := urls.Parse $dest -}} {{- $res := false -}} {{- if $url.Scheme -}} {{- with try (resources.GetRemote $dest) -}} {{- with .Err }}{{ warnf "Image %q in %s could not be fetched: %s" $dest $.Page.String . }}{{ end -}} {{- with .Value }}{{ $res = . }}{{ end -}} {{- end -}} {{- else -}} {{- with resources.Get $dest }}{{ $res = . }}{{ end -}} {{- end -}} {{- define "partials/inline/render-image-img.html" -}} {{- with .res -}} {{- if and (eq .MediaType.SubType "svg") (not $.remote) -}} {{ .Content | safeHTML }} {{- else -}} {{- end -}} {{- else -}} {{- end -}} {{- end -}} {{- $ctx := dict "res" $res "alt" $text "dest" $dest "remote" (ne $url.Scheme "") -}} {{- if .Title -}} {{- else -}} {{ partial "inline/render-image-img.html" $ctx }} {{- end -}}
