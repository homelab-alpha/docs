{{- /* prism shortcode: kept for pages that use it. Prism is gone; the code is highlighted at build time like every code block (layouts/partials/docs/code-block.html). Parameters: lang, line-numbers, start, line (lines to highlight, such as "6-13,15"), line-offset (the number of the first line in "line"), and linkable-line-numbers. Without parameters: a plain block. */ -}} {{- $code := replaceRE "^\r?\n" "" .Inner | string -}} {{- if and .IsNamedParams (gt (len .Params) 0) -}} {{- $options := dict -}} {{- if eq (.Get "line-numbers") "true" }}{{ $options = merge $options (dict "linenos" "table") }}{{ end -}} {{- with .Get "start" }}{{ $options = merge $options (dict "linenostart" (int .)) }}{{ end -}} {{- if eq (.Get "linkable-line-numbers") "true" }}{{ $options = merge $options (dict "anchorlinenos" true) }}{{ end -}} {{- with .Get "line" -}} {{- /* Prism counts the lines from line-offset; Chroma from the first line. */ -}} {{- $offset := sub (int ($.Get "line-offset" | default "1")) 1 -}} {{- $ranges := slice -}} {{- range split . "," -}} {{- $parts := split (trim . " ") "-" -}} {{- $from := sub (int (index $parts 0)) $offset -}} {{- if gt (len $parts) 1 -}} {{- $ranges = $ranges | append (printf "%d-%d" $from (sub (int (index $parts 1)) $offset)) -}} {{- else -}} {{- $ranges = $ranges | append (string $from) -}} {{- end -}} {{- end -}} {{- $options = merge $options (dict "hl_lines" (delimit $ranges " ")) -}} {{- end -}} {{- partial "docs/code-block.html" (dict "code" $code "lang" (.Get "lang") "options" $options) -}} {{- else -}} {{- partial "docs/code-block.html" (dict "code" $code "lang" (.Get 0)) -}} {{- end -}}
